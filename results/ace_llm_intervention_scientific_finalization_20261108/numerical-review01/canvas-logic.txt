function ErrorPoints({world,left,right}:{world:number;left:string;right:string}) {
 const t=useHostTheme();
 const sets=[left,right].map(p=>DATA.curves.filter(x=>x.world_id===world && x.policy_id===p));
 const maximum=Math.max(.05,...sets.flatMap(s=>s.map(p=>p.normalized_mse)))*1.08;
 const x=(v:number)=>70+(v-8)/24*650;
 const y=(v:number)=>260-v/maximum*220;
 return <Stack gap={8}>
  <Row wrap gap={20}><Text><span style={{color:t.accent.primary}}>●</span> {NAMES[left]}</Text><Text><span style={{color:t.text.secondary}}>■</span> {NAMES[right]}</Text></Row>

   <text x={70} y={18} fontSize={12} fill={t.text.secondary}>Normalized mean squared error, lower is better</text>
   {[0,.25,.5,.75,1].map(a=><g key={a}><line x1={70} y1={y(a*maximum)} x2={720} y2={y(a*maximum)} stroke={t.stroke.tertiary}/><text x={60} y={y(a*maximum)+4} textAnchor="end" fontSize={11} fill={t.text.secondary}>{(a*maximum).toFixed(maximum<1?3:2)}</text></g>)}
   {[8,12,16,20,24,28,32].map(v=><g key={v}><line x1={x(v)} y1={40} x2={x(v)} y2={260} stroke={t.stroke.tertiary}/><text x={x(v)} y={280} textAnchor="middle" fontSize={11} fill={t.text.secondary}>{v}</text></g>)}
   <line x1={70} x2={720} y1={y(.04)} y2={y(.04)} stroke={t.text.tertiary} strokeDasharray="5 4"/>
   <text x={725} y={y(.04)+4} fill={t.text.secondary} fontSize={11}>0.04</text>
   {sets[0].map(p=><circle key={p.total_cost_units} cx={x(p.total_cost_units)} cy={y(p.normalized_mse)} r={4.5} fill={t.accent.primary}><title>{NAMES[left]} · cost {p.total_cost_units} · NMSE {p.normalized_mse} · {p.action_id?ACTIONS[p.action_id]:"Shared initialization"}</title></circle>)}
   {sets[1].map(p=><rect key={p.total_cost_units} x={x(p.total_cost_units)-3} y={y(p.normalized_mse)-3} width={6} height={6} fill={t.text.secondary}><title>{NAMES[right]} · cost {p.total_cost_units} · NMSE {p.normalized_mse} · {p.action_id?ACTIONS[p.action_id]:"Shared initialization"}</title></rect>)}

 </Stack>;
}
